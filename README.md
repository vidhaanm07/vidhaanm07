class Vidhaan:

        self.mode = "BUILD"
        self.curiosity = float("inf")

    def learn(self, topic):
        return f"Understanding {topic}..."

    def build(self, idea):
        return f"Turning {idea} into reality..."

    def debug(self, problem):
